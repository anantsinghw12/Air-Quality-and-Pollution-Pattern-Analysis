// Helper: Convert AQI or metric value to color and severity label function getSeverity(val, metric) { let aqiEq = val; if (metric === 'pm25') aqiEq = val * 1.5; else if (metric === 'pm10') aqiEq = val * 0.9; else if (metric === 'no2') aqiEq = val * 2.5; else if (metric === 'co') aqiEq = val * 85; if (aqiEq <= 50) return { label: 'Good', color: 0x34d399, hex: '#34d399', badgeBg: 'bg-emerald-500/20 text-emerald-300 border-emerald-500/40' }; if (aqiEq <= 100) return { label: 'Moderate', color: 0xfacc15, hex: '#facc15', badgeBg: 'bg-yellow-500/20 text-yellow-300 border-yellow-500/40' }; if (aqiEq <= 150) return { label: 'Sensitive', color: 0xf97316, hex: '#f97316', badgeBg: 'bg-orange-500/20 text-orange-300 border-orange-500/40' }; if (aqiEq <= 250) return { label: 'Unhealthy', color: 0xef4444, hex: '#ef4444', badgeBg: 'bg-red-500/20 text-red-300 border-red-500/40' }; return { label: 'Hazardous', color: 0xa855f7, hex: '#a855f7', badgeBg: 'bg-purple-500/20 text-purple-300 border-purple-500/40' }; } // ========================================================================= // 2.
